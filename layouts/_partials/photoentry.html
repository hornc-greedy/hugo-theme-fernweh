"url" .RelPermalink "caption" $caption "w" .Width "h" .Height "date" "" "small" $small.RelPermalink "sw" $small.Width "sh" $small.Height "medium" $medium.RelPermalink "mw" $medium.Width "mh" $medium.Height "pin" (.Fill "128x128 center").RelPermalink "pinlarge" (.Fill "440x440 center").RelPermalink -}} {{- /* a camera writes the moment and the place into the file; a scan carries neither, and then the album falls back to the order of the folder */ -}} {{- if reflect.IsImageResourceWithMeta . -}} {{- with .Meta -}} {{- $entry = merge $entry (dict "date" (.Date.Format "2006-01-02T15:04:05")) -}} {{- if .Lat -}} {{- $entry = merge $entry (dict "lat" .Lat "lon" .Long) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $entry -}}
