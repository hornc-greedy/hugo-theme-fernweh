{{- /* The maps of a journey, the one with the most photos first. A photo belongs to the part of a country it lies in, or, on an island too small for the outlines, to the nearest part within a degree. Parts of one country whose photos lie less than 100 km apart share a map, so a hop between islands stays on one while Sardinia keeps its own. A part less than 40 km across cannot be told apart on its own and draws in every part of its country within 100 km, which puts Norderney on the map of Germany. A smaller part without photos that reaches into the box of a map joins the smallest such map, so Sicily stays whole beside the mainland while Java stays off the map of Bali, and a photo off every coast, from a boat or a plane, joins the map whose box holds it most closely. Each map names the cells of its photos, which is how the caller sets every photo on exactly one map. The answer depends on the coordinates alone and not on the language of the page, which is why the caller asks for it through partialCached. */ -}} {{- $cells := .cells -}} {{- $features := (resources.Get "geo/countries.json" | transform.Unmarshal).features -}} {{- /* ── the part each photo lies in or next to ────────────────────────── */ -}} {{- $matches := slice -}} {{- $unmatched := slice -}} {{- range $cell := $cells -}} {{- $match := dict -}} {{- range $c, $country := $features -}} {{- $b := $country.properties.box -}} {{- if and (not $match) (ge $cell.lat (index $b 0)) (le $cell.lat (index $b 2)) (ge $cell.long (index $b 1)) (le $cell.long (index $b 3)) -}} {{- range $i, $part := $country.geometry.coordinates -}} {{- $pb := index $country.properties.boxes $i -}} {{- /* the outline is only walked for a point that its own box holds; an island on the far side of the country costs four comparisons */ -}} {{- if and (not $match) (ge $cell.lat (index $pb 0)) (le $cell.lat (index $pb 2)) (ge $cell.long (index $pb 1)) (le $cell.long (index $pb 3)) -}} {{- if partial "inside-part.html" (dict "lat" $cell.lat "long" $cell.long "part" $part) -}} {{- $match = dict "country" $c "part" $i -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $match -}} {{- $cos := math.Cos (mul $cell.lat 0.0174533) -}} {{- /* squared kilometres, a degree of latitude at the most */ -}} {{- $minDistance := 12365.0 -}} {{- range $c, $country := $features -}} {{- range $i, $pb := $country.properties.boxes -}} {{- if and (ge $cell.lat (sub (index $pb 0) 1)) (le $cell.lat (add (index $pb 2) 1)) (ge $cell.long (sub (index $pb 1) 1)) (le $cell.long (add (index $pb 3) 1)) -}} {{- range index (index $country.geometry.coordinates $i) 0 -}} {{- $dy := mul (sub (index . 1) $cell.lat) 111.2 -}} {{- $dx := mul (mul (sub (index . 0) $cell.long) 111.2) $cos -}} {{- $d := add (mul $dx $dx) (mul $dy $dy) -}} {{- if lt $d $minDistance -}} {{- $minDistance = $d -}} {{- $match = dict "country" $c "part" $i -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $match -}} {{- $matches = $matches | append (merge $match (dict "cell" $cell)) -}} {{- else -}} {{- $unmatched = $unmatched | append $cell -}} {{- end -}} {{- end -}} {{- /* ── the parts each photo draws in, gathered into maps ─────────────── */ -}} {{- $clusters := slice -}} {{- range $matches -}} {{- $cell := .cell -}} {{- $country := index $features .country -}} {{- $cos := math.Cos (mul $cell.lat 0.0174533) -}} {{- $radiusLat := 0.9 -}} {{- $radiusLong := div 0.9 $cos -}} {{- -}} {{- range $g.cells -}}{{- $photos = add $photos .count -}}{{- end -}} {{- /* the numbers are collected, not the outlines: append unwraps a slice, and a second outline cannot be appended to a list
