{{- /* The dominant colour of the covers a page shows: its own on an entry, and on a list the covers of the entries listed, averaged. How strongly that colour is applied is settled in the stylesheet, not here. The colours are hex values, which int reads with 0x in front. */ -}} {{- $imgs := slice -}} {{- with partial "cover.html" . -}} {{- $imgs = $imgs | append . -}} {{- else -}} {{- range .Pages -}} {{- $c := partial "cover.html" . -}} {{- if not $c -}} {{- with .Pages.ByDate -}}{{ $c = partial "cover.html" (index . 0) }}{{- end -}} {{- end -}} {{- with $c -}}{{ $imgs = $imgs | append . }}{{- end -}} {{- end -}} {{- end -}} {{- $r := 0.0 -}} {{- $g := 0.0 -}} {{- $b := 0.0 -}} {{- $n := 0 -}} {{- range $imgs -}} {{- with .color -}} {{- $r = add $r (int (printf "0x%s" (substr . 1 2))) -}} {{- $g = add $g (int (printf "0x%s" (substr . 3 2))) -}} {{- $b = add $b (int (printf "0x%s" (substr . 5 2))) -}} {{- $n = add $n 1 -}} {{- end -}} {{- end -}} {{- $hex := "" -}} {{- if $n -}} {{- $hex = printf "#%02x%02x%02x" (int (div $r $n)) (int (div $g $n)) (int (div $b $n)) -}} {{- end -}} {{- return $hex -}}
