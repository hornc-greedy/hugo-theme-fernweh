{{- /* The meta description of a page is its own, or else how many photographs the entry holds and when they were taken. Empty for a page with neither, so the site's own description stays the fallback. */ -}} {{- $text := .Description -}} {{- if not $text -}} {{- $count := 0 -}} {{- $span := "" -}} {{- if .IsPage -}} {{- $count = len (partialCached "photos.html" . .Path) -}} {{- $span = partial "entrydate.html" . -}} {{- else if .IsSection -}} {{- range .Pages -}} {{- $count = add $count (len (partialCached "photos.html" . .Path)) -}} {{- end -}} {{- $span = partial "daterange.html" . -}} {{- end -}} {{- if $count -}} {{- $text = printf "%s · %s" (i18n "photographs" $count) (plainify $span | strings.TrimSpace) -}} {{- end -}} {{- end -}} {{- return $text -}}
