{{- /* A boundary as the map compares it: the feature, the code of its country if it is a state, its area for the order, scaled by `weight`, and its longer side for the margin, both in kilometres. */ -}} {{- $b := .feature.properties.box -}} {{- $cos := math.Cos (math.ToRadians (div (add (index $b 0) (index $b 2)) 2)) -}} {{- $height := mul (sub (index $b 2) (index $b 0)) 111.2 -}} {{- $width := mul (sub (index $b 3) (index $b 1)) 111.2 $cos -}} {{- return dict "feature" .feature "country" (.feature.properties.country | default "") "area" (mul $height $width .weight) "size" (math.Max $height $width) -}}
