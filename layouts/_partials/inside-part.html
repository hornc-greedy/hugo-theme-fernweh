{{- /* Whether a point lies inside a part. Ray casting counts how often a ray to the east crosses its rings, and an odd count means inside, so a point in a hole lies outside. */ -}} {{- $x := .long -}} {{- $y := .lat -}} {{- $inside := false -}} {{- range $ring := .part -}} {{- range $i, $p := after 1 $ring -}} {{- $q := index $ring $i -}} {{- $xi := index $p 0 -}} {{- $yi := index $p 1 -}} {{- $xj := index $q 0 -}} {{- $yj := index $q 1 -}} {{- if ne (gt $yi $y) (gt $yj $y) -}} {{- if lt $x (add (div (mul (sub $xj $xi) (sub $y $yi)) (sub $yj $yi)) $xi) -}} {{- $inside = not $inside -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $inside -}}
