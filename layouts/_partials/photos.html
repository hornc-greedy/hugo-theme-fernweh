{{- /* The photos of a day. With a bucket configured they are the selection from the front matter; without one they are the pictures lying beside the entry. Either way in the order they were taken. */ -}} {{- $page := . -}} {{- $images := slice -}} {{- if site.Params.photos.origin -}} {{- range .Params.photos -}} {{- $images = $images | append (partial "photoentry.html" (dict "url" .image "caption" .caption)) -}} {{- end -}} {{- else -}} {{- $captions := dict -}} {{- range .Params.photos -}} {{- $captions = merge $captions (dict .image (.caption | default "")) -}} {{- end -}} {{- range $page.Resources.ByType "image" -}} {{- $images = $images | append (partial "photoentry.html" (dict "file" . "caption" (index $captions .Name))) -}} {{- end -}} {{- end -}} {{- return (sort $images "date") -}}
