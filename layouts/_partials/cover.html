{{- /* The opener of an entry: the picked cover, or the first of its photos. */ -}} {{- $page := . -}} {{- $cover := "" -}} {{- with .Params.cover -}} {{- $name := . -}} {{- /* the caption of a picked cover stands beside it in the list of photos, which a journey does not have */ -}} {{- $caption := "" -}} {{- with $page.Params.photos -}} {{- with where . "image" $name -}}{{- $caption = (index . 0).caption -}}{{- end -}} {{- end -}} {{- if site.Params.photos.origin -}} {{- $cover = partial "photoentry.html" (dict "url" $name "caption" $caption) -}} {{- else -}} {{- with $page.Resources.GetMatch $name -}} {{- $cover = partial "photoentry.html" (dict "file" . "caption" $caption) -}} {{- else -}} {{- errorf "cover missing from the entry: %s in %s" $name $page.File.Path -}} {{- end -}} {{- end -}} {{- else -}} {{- with partialCached "photos.html" . .Path -}} {{- $cover = index . 0 -}} {{- end -}} {{- end -}} {{- return $cover -}}
