{{- /* Splits pictures into rows of about equal height. The number of rows comes from the total shape of the album; each row is then filled to the average of what is still unplaced, and closed where that average is nearest, so the last row is as full as the others instead of holding the remainder. Takes the pictures, the shape a row should have and the fewest pictures a row may hold; returns every row with its shape. */ -}} {{- $items := .items -}} {{- $least := .least -}} {{- $n := mul 1.0 (len $items) -}} {{- $sum := 0.0 -}} {{- range $items }}{{ $sum = add $sum .ar }}{{ end -}} {{- $count := math.Max 1.0 (math.Min (math.Round (div $sum .shape)) (math.Floor (div $n $least))) -}} {{- $rows := slice -}} {{- $row := slice -}} {{- $acc := 0.0 -}} {{- $placed := 0.0 -}} {{- range $i, $p := $items -}} {{- $open := sub $count (mul 1.0 (len $rows)) -}} {{- $after := sub $open 1.0 -}} {{- $target := div (sub $sum $placed) $open -}} {{- $rest := sub $n (mul 1.0 $i) -}} {{- $keep := mul $after $least -}} {{- $near := lt (sub $target $acc) (sub (add $acc $p.ar) $target) -}} {{- if and (gt $after 0.0) (ge (mul 1.0 (len $row)) $least) (ge $rest $keep) (or $near (lt (sub $rest 1.0) $keep)) -}} {{- $rows = $rows | append (dict "items" $row "sum" $acc) -}} {{- $placed = add $placed $acc -}} {{- $row = slice -}} {{- $acc = 0.0 -}} {{- end -}} {{- $row = $row | append $p -}} {{- $acc = add $acc $p.ar -}} {{- end -}} {{- $rows = $rows | append (dict "items" $row "sum" $acc) -}} {{- return $rows -}}
